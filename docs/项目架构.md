### 1.2 核心设计原则

#### 1.2.1 微服务原则
- **单一职责**: 每个服务只负责一个业务领域
- **自治性**: 服务独立部署、独立扩展
- **去中心化**: 避免单点故障
- **故障隔离**: 服务失败不影响其他服务

#### 1.2.2 松耦合设计
- **接口标准化**: 统一的 RESTful API 规范
- **消息驱动**: 异步通信降低耦合
- **配置外部化**: 配置与代码分离
- **依赖倒置**: 面向接口编程

#### 1.2.3 可扩展性
- **水平扩展**: 支持实例动态增减
- **插件化架构**: 策略模式实现功能扩展
- **版本控制**: API 版本管理
- **向后兼容**: 升级不影响现有功能

**2. 热度评分算法**
```python
def calculate_trending_score(item: TrendingItem) -> float:
    """
    热度评分算法
    Score = (热度值 * 时间衰减系数) + (增长率 * 10) + (质量分 * 5)
    """
    time_decay = math.exp(-0.1 * item.hours_since_publish)
    growth_rate = item.current_heat / max(item.initial_heat, 1)
    quality_score = item.engagement_rate * 10
    
    return (item.heat_value * time_decay) + (growth_rate * 10) + (quality_score * 5)
```

#### 2.3.3 视频后期处理流程
```python
class VideoPostProcessor:
    """视频后期处理器"""
    
    async def process(self, video: Video, script: Script) -> Video:
        """
        处理流程:
        1. 添加字幕
        2. 混音（背景音乐 + 旁白）
        3. 添加转场特效
        4. 添加水印/Logo
        5. 格式转换
        6. 质量检测
        """
        video = await self._add_subtitles(video, script.subtitles)
        video = await self._mix_audio(video, script.audio_config)
        video = await self._add_transitions(video, script.transitions)
        video = await self._add_watermark(video)
        video = await self._convert_format(video, target_format="mp4")
        
        quality_check = await self._quality_check(video)
        if not quality_check.passed:
            raise VideoQualityError(quality_check.issues)
        
        return video
```

#### 2.4.2 工作流 DAG 定义
```python
class WorkflowDAG:
    """工作流 DAG"""
    
    def __init__(self, name: str):
        self.name = name
        self.tasks: Dict[str, Task] = {}
        self.dependencies: Dict[str, List[str]] = {}
    
    def add_task(self, task: Task, depends_on: List[str] = None):
        """添加任务节点"""
        self.tasks[task.id] = task
        self.dependencies[task.id] = depends_on or []
    
    def execute(self, context: dict):
        """执行工作流"""
        execution_order = self._topological_sort()
        
        for task_id in execution_order:
            task = self.tasks[task_id]
            result = await self._execute_task(task, context)
            context[task_id] = result

# 使用示例
workflow = WorkflowDAG("auto_video_generation")

workflow.add_task(
    Task(id="fetch_trending", func=fetch_trending_topics, timeout=300)
)

workflow.add_task(
    Task(id="generate_script", func=generate_script, timeout=600),
    depends_on=["fetch_trending"]
)

workflow.add_task(
    Task(id="create_video", func=create_video, timeout=1800),
    depends_on=["generate_script"]
)
```

        retry_delay = 60 * (2 ** self.request.retries)

#### 3.2.2 缓存更新策略

class AuthService:
    """认证服务"""
    
    def generate_token(self, user_id: str) -> str:
        """生成 JWT Token"""
        payload = {
            'user_id': user_id,
            'exp': datetime.utcnow() + timedelta(hours=24),
            'iat': datetime.utcnow()
        }
        return jwt.encode(payload, settings.SECRET_KEY, algorithm='HS256')
    
    def verify_token(self, token: str) -> Optional[dict]:
        """验证 Token"""
        try:
            payload = jwt.decode(token, settings.SECRET_KEY, algorithms=['HS256'])
            return payload
        except jwt.ExpiredSignatureError:
            raise AuthenticationError("Token expired")
        except jwt.InvalidTokenError:
            raise AuthenticationError("Invalid token")
```

class SlidingWindowRateLimiter:
    async def is_allowed(self, key: str, limit: int, window: int) -> bool:
        now = time.time()
        window_start = now - window
        
        # 清除过期记录
        await redis.zremrangebyscore(key, 0, window_start)
        
        # 检查当前窗口内的请求数
        count = await redis.zcard(key)
        if count >= limit:
            return False
        
        # 记录当前请求
        await redis.zadd(key, {str(uuid.uuid4()): now})
        await redis.expire(key, window)
        
        return True
```

async def batch_generate_videos(trending_items: List[TrendingItem]):
    """批量生成视频"""
    
    # 批量生成剧本（并发）
    script_tasks = [
        generate_script_async(item) 
        for item in trending_items
    ]
    scripts = await asyncio.gather(*script_tasks)
    
    # 批量生成视频（限制并发数）
    semaphore = asyncio.Semaphore(5)  # 最多5个并发
    
    async def generate_with_limit(script):
        async with semaphore:
            return await generate_video_async(script)
    
    video_tasks = [generate_with_limit(script) for script in scripts]
    videos = await asyncio.gather(*video_tasks)
    
    return videos
